#+begin_src elisp
  (defcustom gdbprog "/usr/bin/gdb"
    "Path to gdb tool."
    :type 'string
    :group 'your-package-name)
  (defcustom gdbport "-ex \"target remote :50010\""
    "gdb server port"
    :type 'string
    :group 'your-package-name)
  (defcustom debuggerpath "/path/to/la_dbg_tool_usb"
    "Path to the EJTAG debugger tool."
    :type 'string
    :group 'your-package-name)
  (defcustom  ldbcfg "source configs/config.ls2k"
    "board config"
    :type 'string
    :group 'your-package-name)
  (defcustom ldbelf "pmon.gdb"
    "elf file for gdb to debug."
    :type 'string
    :group 'your-package-name)

  (defcustom ldbrom "zloader/gzrom.bin"
    "elf file for gdb to debug."
    :type 'string
    :group 'your-package-name)

#+end_src
